pythonfrom prometheus_fastapi_instrumentator import Instrumentator
from prometheus_client import Counter, Histogram, Gauge
import time

# --- Define your custom metrics ---
IMAGES_GENERATED = Counter(
    "genai_images_generated_total",
    "Total images generated",
    ["model_mode"]          # label: baseline / light / overfit
)

INFERENCE_LATENCY = Histogram(
    "genai_inference_latency_seconds",
    "End-to-end inference latency",
    ["model_mode"],
    buckets=[2, 4, 6, 8, 10, 15, 20]   # seconds — tuned for SD
)

CLIP_SCORE = Histogram(
    "genai_clip_score",
    "CLIP score of generated output",
    ["model_mode"],
    buckets=[5, 10, 15, 20, 25, 28, 30]
)

GPU_VRAM_USAGE = Gauge(
    "genai_gpu_vram_bytes",
    "Current GPU VRAM usage in bytes"
)

API_ERRORS = Counter(
    "genai_api_errors_total",
    "Total API errors",
    ["error_type"]          # label: rembg_failed / oom / timeout
)

# --- Auto-instrument all HTTP routes (request count, latency) ---
app = FastAPI()
Instrumentator().instrument(app).expose(app)   # exposes /metrics endpoint

# --- Update VRAM gauge in background ---
import torch, threading

def poll_vram():
    while True:
        if torch.cuda.is_available():
            used = torch.cuda.memory_allocated(0)
            GPU_VRAM_USAGE.set(used)
        time.sleep(5)

threading.Thread(target=poll_vram, daemon=True).start()
Step 3 — instrument your /generate endpoint
python@app.post("/generate")
async def generate(image: UploadFile, prompt: str, model_mode: str = "baseline", ...):
    start = time.time()
    try:
        # --- your existing rembg + SD inpainting logic here ---
        result_image, clip_score = run_pipeline(image, prompt, model_mode)

        # Record metrics
        duration = time.time() - start
        IMAGES_GENERATED.labels(model_mode=model_mode).inc()
        INFERENCE_LATENCY.labels(model_mode=model_mode).observe(duration)
        CLIP_SCORE.labels(model_mode=model_mode).observe(clip_score)

        return result_image

    except torch.cuda.OutOfMemoryError:
        API_ERRORS.labels(error_type="oom").inc()
        raise HTTPException(503, "GPU out of memory")
    except Exception as e:
        API_ERRORS.labels(error_type="rembg_failed").inc()
        raise HTTPException(500, str(e))
